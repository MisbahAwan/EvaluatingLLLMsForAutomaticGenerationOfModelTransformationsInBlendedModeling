  <define keyword="#define" name="DESIRED_SPEED_INCREMENT" value="1"

  <define keyword="#define" name="MAX_SPEED" value="150"
          comment="Maximum allowed speed (in km/h)"/>
  <define keyword="#define" name="MIN_SPEED" value="30"
          comment="Minimum allowed speed (in km/h)"/>
  <define keyword="#define" name="THROTTLE_MAX" value="100"
          comment="Maximum throttle value (percentage)"/>
  <define keyword="#define" name="THROTTLE_MIN" value="0"
          comment="Minimum throttle value (percentage)"/>

  <!-- Enum type -->
  <enum name="CruiseState" comment="Enum to represent the system state">
    <item name="CRUISE_OFF"/>
    <item name="CRUISE_ON"/>
    <item name="CRUISE_HOLD"/>
  </enum>

  <variable name="desiredSpeed"
            dataType="int"
            isGlobal="true"

            comment="The speed at which cruise control should maintain (km/h)"/>

  <variable name="throttle"
            dataType="int"
            isGlobal="true"
            initialValue="0"
            comment="Throttle level (0 to 100%)"/>

  <variable name="state"
            dataType="CruiseState"
            isGlobal="true"
            initialValue="CRUISE_OFF"
            comment="Initial state is OFF"/>

  <!-- Function: controlThrottle -->
  <function name="controlThrottle"
            returnType="void"
            comment="Function to control the throttle">
    <parameter name="speedDifference" dataType="int"/>

    <body>
      <if condition="speedDifference &gt; 0">
        <then>
          <assign left="throttle" operator="+=" right="speedDifference"/>
          <comment text="Increase throttle to speed up"/>
        </then>
        <elseif condition="speedDifference &lt; 0">
          <then>
            <assign left="throttle" operator="+=" right="speedDifference"/>
            <comment text="Decrease throttle to slow down"/>
          </then>
        </elseif>
      </if>

      <comment text="Clamp throttle to valid range (0% to 100%)"/>

      <if condition="throttle &gt; THROTTLE_MAX">
        <then>
          <assign left="throttle" operator="=" right="THROTTLE_MAX"/>
        </then>
      </if>

      <if condition="throttle &lt; THROTTLE_MIN">
        <then>
          <assign left="throttle" operator="=" right="THROTTLE_MIN"/>
        </then>
      </if>

      <print expression="printf(&quot;Throttle set to: %d%%\n&quot;, throttle);"/>
    </body>
  </function>

  <!-- Function: updateCruiseControl -->
  <function name="updateCruiseControl"
            returnType="void"
            comment="Function to update the cruise control system state">

    <parameter name="setButtonPressed" dataType="int"/>
    <parameter name="increaseSpeed" dataType="int"/>
    <parameter name="decreaseSpeed" dataType="int"/>
    <parameter name="cancelButtonPressed" dataType="int"/>

    <local dataType="int" name="speedDifference"/>

    <body>
      <switch expression="state">

        <!-- case CRUISE_OFF -->
        <case value="CRUISE_OFF">
          <if condition="setButtonPressed &amp;&amp; currentSpeed &gt;= MIN_SPEED">
            <then>
              <assign left="desiredSpeed" operator="=" right="currentSpeed"/>
              <assign left="state" operator="=" right="CRUISE_ON"/>
              <print expression="printf(&quot;Cruise Control ON. Holding speed at %d km/h\n&quot;, desiredSpeed);"/>
            </then>
          </if>
          <break/>
        </case>

        <!-- case CRUISE_ON -->
        <case value="CRUISE_ON">
          <if condition="cancelButtonPressed">
            <then>
              <assign left="state" operator="=" right="CRUISE_OFF"/>
              <assign left="throttle" operator="=" right="0"/>
              <print expression="printf(&quot;Cruise Control OFF\n&quot;);"/>
            </then>

            <elseif condition="increaseSpeed">
              <then>
                <assign left="desiredSpeed" operator="+=" right="DESIRED_SPEED_INCREMENT"/>

                <if condition="desiredSpeed &gt; MAX_SPEED">
                  <then>
                    <assign left="desiredSpeed" operator="=" right="MAX_SPEED"/>
                  </then>
                </if>

                <print expression="printf(&quot;Increasing desired speed to %d km/h\n&quot;, desiredSpeed);"/>
              </then>
            </elseif>

            <elseif condition="decreaseSpeed">
              <then>
                <assign left="desiredSpeed" operator="-=" right="DESIRED_SPEED_INCREMENT"/>

                <if condition="desiredSpeed &lt; MIN_SPEED">
                  <then>
                    <assign left="desiredSpeed" operator="=" right="MIN_SPEED"/>
                  </then>
                </if>

                <print expression="printf(&quot;Decreasing desired speed to %d km/h\n&quot;, desiredSpeed);"/>
              </then>
            </elseif>

            <else>
              <assign left="state" operator="=" right="CRUISE_HOLD"/>
            </else>
          </if>

          <break/>
        </case>

        <!-- case CRUISE_HOLD -->
        <case value="CRUISE_HOLD">
          <if condition="cancelButtonPressed">
            <then>
              <assign left="state" operator="=" right="CRUISE_OFF"/>
              <assign left="throttle" operator="=" right="0"/>
              <print expression="printf(&quot;Cruise Control OFF\n&quot;);"/>
            </then>

            <else>
              <comment text="Control the throttle to maintain the desired speed"/>
              <assign left="speedDifference" operator="=" right="desiredSpeed - currentSpeed"/>
              <call functionName="controlThrottle" arguments="speedDifference"/>
            </else>
          </if>

          <break/>
        </case>

      </switch>
    </body>
  </function>

    </body>
  </function>
